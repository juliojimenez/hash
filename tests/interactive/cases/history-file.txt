# Typed commands reach the history file as typed, each behind its #<seconds> marker;
# a leading space keeps a line out (HISTCONTROL default)
send echo FIRST''-CMD
expect FIRST-CMD
send echo SECOND''-CMD
expect SECOND-CMD
send  echo SPA''CED
expect SPACED
send exit
status 0
file .hash_history has echo FIRST''-CMD\n#
file .hash_history has echo SECOND''-CMD\n#
file .hash_history lacks SPACED
file .hash_history has exit
